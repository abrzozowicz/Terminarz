async function save() { const descInput = document.getElementById('fDesc'); const dateInput = document.getElementById('fDate'); const timeInput = document.getElementById('fTime'); if(!descInput.value) return alert("Wpisz nazwę wydarzenia!"); const ev = { id: document.getElementById('eId').value || Date.now(), d: dateInput.value, t: timeInput.value, desc: descInput.value }; // 1. Aktualizacja lokalna (żebyś widział zmiany od razu) const idx = db.findIndex(x => x.id == ev.id); if(idx > -1) db[idx] = ev; else db.push(ev); render(); descInput.value = ""; // 2. Przygotowanie przycisku const btn = document.getElementById('saveBtn'); btn.innerText = "SYNCHRONIZACJA..."; // 3. WYSYŁKA - zmieniona metoda na przesyłanie tekstu jako parametr try { await fetch(SCRIPT_URL, { method: 'POST', body: JSON.stringify({ action: 'save', events: db }), headers: { 'Content-Type': 'text/plain;charset=utf-8' } // To omija błędy CORS }); btn.innerText = "ZAPISANO!"; setTimeout(() => { btn.innerText = "Zapisz w Arkuszu"; }, 2000); } catch (e) { console.error("Błąd wysyłki:", e); btn.innerText = "BŁĄD POŁĄCZENIA"; } }
